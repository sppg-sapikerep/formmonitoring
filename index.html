Kode Script Google Apps Script (Salin ke Extensions > Apps Script): 
function doGet(e) {
  try {
    var folderId = "1OxQd1lFN1wq1T7-obgOiHrjPTKWTv2Hy";
    var folder = DriveApp.getFolderById(folderId);
    var dbFile = folder.getFilesByName("db_sppg_records.json");
    var records = [];
    if (dbFile.hasNext()) {
      var content = dbFile.next().getBlob().getDataAsString();
      records = JSON.parse(content);
    }
    return ContentService.createTextOutput(JSON.stringify({status: "success", data: records}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var folderId = "1OxQd1lFN1wq1T7-obgOiHrjPTKWTv2Hy";
    var folder = DriveApp.getFolderById(folderId);
    var fileUrl = "";
    
    if (data.pdfBase64) {
      var bytes = Utilities.base64Decode(data.pdfBase64);
      var blob = Utilities.newBlob(bytes, "application/pdf", data.fileName);
      var file = folder.createFile(blob);
      fileUrl = file.getUrl();
    }
    
    if (data.record) {
      data.record.driveUrl = fileUrl || data.record.driveUrl || "";
      var dbFile = folder.getFilesByName("db_sppg_records.json");
      var records = [];
      var dbFileObj = null;
      if (dbFile.hasNext()) {
        dbFileObj = dbFile.next();
        try { records = JSON.parse(dbFileObj.getBlob().getDataAsString()); } catch(ex){}
      }
      
      var idx = records.findIndex(function(r){ return r.id === data.record.id; });
      if (idx >= 0) { records[idx] = data.record; } else { records.unshift(data.record); }
      
      if (dbFileObj) {
        dbFileObj.setContent(JSON.stringify(records));
      } else {
        folder.createFile("db_sppg_records.json", JSON.stringify(records), "application/json");
      }
    }
    
    return ContentService.createTextOutput(JSON.stringify({status: "success", fileUrl: fileUrl}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({status: "error", message: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
